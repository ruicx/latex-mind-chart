\newcommand{\checkPublicationNode}[2]{%
  \pgfpointdiff{\pgfpointanchor{mc_node_#1}{west}}{\pgfpointanchor{mc_node_#1}{east}}%
  \directlua{assert(math.abs(\number\pgf@x/65536-#2)<0.1, 'styled rendered width disagrees with layout')}%
}
\newcommand{\checkPublicationBounds}{%
  \directlua{
    for _, name in ipairs({'a','n'}) do
      local n = require('mindchart.ast').named_nodes[name]
      tex.sprint(string.char(92)..'checkPublicationNode{'..n.id..'}{'..n.box_width..'}')
    end
  }%
}
\makeatother
